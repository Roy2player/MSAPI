add_compile_options(
    -Wall
    -Wno-unknown-pragmas
    -Wextra
    -Wpedantic

    -Wcast-align
    -Wcast-qual
    -Wconversion
    -Wctor-dtor-privacy
    -Wenum-compare
    -Wfloat-equal
    -Wnon-virtual-dtor
    -Wold-style-cast
    -Woverloaded-virtual
    -Wredundant-decls

    -Wsign-promo

    #-Werror

    -std=c++26

    -fconcepts-diagnostics-depth=2

    -fdiagnostics-color=always
)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(STATUS "Build Type: Debug")
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    message(STATUS "Build Type: Release")
else()
    message(STATUS "Build Type: Unknown")
endif()

    message(STATUS "C++ Compile Options: ${CMAKE_CXX_FLAGS}")
endif()

# Check if CMAKE_C_FLAGS is defined and not empty
if(DEFINED CMAKE_C_FLAGS AND NOT CMAKE_C_FLAGS STREQUAL "")
    message(STATUS "C Compile Options: ${CMAKE_C_FLAGS}")
endif()

# Check if CMAKE_CXX_DEFINES (compile definitions) is defined and not empty
if(DEFINED CMAKE_CXX_DEFINES AND NOT CMAKE_CXX_DEFINES STREQUAL "")
    message(STATUS "Compile Definitions: ${CMAKE_CXX_DEFINES}")
endif()

# Check if the project is being compiled in 32-bit or 64-bit mode
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(STATUS "Data Model: 64-bit")
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(STATUS "Data Model: 32-bit")
else()
    message(STATUS "Unknown Data Model")
endif()

# Print the compiler version and information
message(STATUS "Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "Compiler Target: ${CMAKE_SYSTEM_PROCESSOR}")

set(MSAPI_OPTIMIZE_OPTIONS
    -O3 # TODO: Find a way to build with -Ofast
    -march=native
    -flto=auto
    -funroll-loops
    -finline-functions
    -fomit-frame-pointer
)

# UndefinedBehaviorSanitizer includes signed overflow, shift, vla-bound, null,
# bounds, bool, enum, vptr, and pointer-overflow checks. Integer divide-by-zero
# is listed explicitly.
set(MSAPI_SANITIZE_OPTIONS
    -fsanitize=address,leak,undefined,integer-divide-by-zero,float-divide-by-zero,float-cast-overflow
    -fno-sanitize-recover=all
)

# Keep build profiles in the existing build-type conditionals.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    # Debug: modest optimization, full debug symbols, frame pointers, assertions, sanitizers.
    add_compile_options(
        -Og
        -g3
        -v
        -fno-omit-frame-pointer
        -D_GLIBCXX_ASSERTIONS
        ${MSAPI_SANITIZE_OPTIONS}
    )
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    # Release: optimized build with LTO.
    add_compile_options(${MSAPI_OPTIMIZE_OPTIONS})
endif()

# Report compile options after adding the selected build profile.
get_property(CXX_COMPILE_OPTIONS DIRECTORY PROPERTY COMPILE_OPTIONS)
if(CXX_COMPILE_OPTIONS)
    message(STATUS "C++ Additional Compile Options: ${CXX_COMPILE_OPTIONS}")
endif()

# Function to set common properties to MSAPI application target
function(msapi_set_target_properties target)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD_REQUIRED YES
        CXX_EXTENSIONS NO
    )

    target_link_options(${target} PRIVATE -static-libstdc++ -static-libgcc)

    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        target_link_options(${target} PRIVATE ${MSAPI_SANITIZE_OPTIONS})
    elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
        target_link_options(${target} PRIVATE -flto=auto)
    endif()
endfunction()
